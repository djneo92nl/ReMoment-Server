  fk (artist_id) -> artists(id) ON DELETE CASCADE ON UPDATE NO ACTION

TABLE users [autoincrement]
  col created_at datetime NULL default=-
  col email varchar NULL default=-
  col email_verified_at datetime NULL default=-
  col id integer NOT NULL default=- pk=1
  col name varchar NOT NULL default=-
  col password varchar NOT NULL default=-
  col remember_token varchar NULL default=-
  col updated_at datetime NULL default=-
  idx users_email_unique UNIQUE (email)

TABLE wanted_tracks [autoincrement]
  col album varchar NULL default=-
  col artist varchar NOT NULL default=-
  col artist_key varchar NOT NULL default=-
  col created_at datetime NULL default=-
  col fulfilled_at datetime NULL default=-
  col id integer NOT NULL default=- pk=1
  col image_url text NULL default=-
  col source varchar NOT NULL default='manual'
  col title varchar NOT NULL default=-
  col title_key varchar NOT NULL default=-
  col track_id integer NULL default=-
  col updated_at datetime NULL default=-
  col wanted_at datetime NOT NULL default=-
  idx wanted_tracks_artist_key_title_key_unique UNIQUE (artist_key,title_key)
  idx wanted_tracks_track_id_index INDEX (track_id)
  idx wanted_tracks_wanted_at_index INDEX (wanted_at)
  fk (track_id) -> tracks(id) ON DELETE SET NULL ON UPDATE NO ACTION
